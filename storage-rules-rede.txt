REGRAS DO FIREBASE STORAGE — Rede Liberdade 2.0
================================================
Onde: Firebase Console → Storage → Regras → substitua TUDO pelo bloco abaixo
e clique em Publicar. (Já inclui as fotos de perfil/inscrição em fotos_alunos/
e a pasta apresentacoes/<uid>/ dos vídeos de cada Instrutor/Professor/Mestre.)

ATENÇÃO: a regra de apresentacoes/ consulta o seu cadastro no Firestore para
saber se você é Admin/Fundador. Na primeira vez o Console pode mostrar um aviso
pedindo para "conceder permissão" do Storage ao Firestore — clique em Conceder.

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    function perfilDe(id) { return firestore.get(/databases/(default)/documents/usuarios/$(id)).data; }
    function podeMexerNaApresentacao(uid) {
      return 'admin' in perfilDe(request.auth.uid).get('papeis', [])
          || perfilDe(request.auth.uid).get('acessoGeral', false) == true
          || (request.auth.uid == uid && (
                perfilDe(request.auth.uid).get('cordaoAtual', '') in ['Instrutor', 'Professor', 'Mestre', 'Mestre/Presidente']
                || perfilDe(request.auth.uid).get('papeis', []).hasAny(['mestre', 'instrutor'])))
          || ('mestre' in perfilDe(request.auth.uid).get('papeis', [])
              && perfilDe(request.auth.uid).get('academiaGerenciadaId', null) != null
              && perfilDe(uid).get('academiaId', '') == perfilDe(request.auth.uid).get('academiaGerenciadaId', null));
    }

    // Foto de perfil/carteirinha (inscrição, painel e Rede): qualquer pessoa
    // logada grava (imagem até 3 MB); leitura pública (aparece no app e no site).
    match /fotos_alunos/{arquivo} {
      allow read: if true;
      allow write: if request.auth != null
                   && request.resource.contentType.matches('image/.*')
                   && request.resource.size < 3 * 1024 * 1024;
    }

    // Rede Liberdade: cada pessoa grava só na própria pasta rede/<uid>/.
    // Fotos já chegam comprimidas pelo app (~350 KB) — o limite de 2 MB é só
    // a trava de segurança. Vídeos curtos (até 20 s no app) até 12 MB.
    // Leitura só pra quem está logado (rede fechada).
    match /rede/{uid}/{arquivo} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
                   && request.auth.uid == uid
                   && (
                     (request.resource.contentType.matches('image/.*') && request.resource.size < 2 * 1024 * 1024) ||
                     (request.resource.contentType.matches('video/.*') && request.resource.size < 12 * 1024 * 1024)
                   );
      allow delete: if request.auth != null && request.auth.uid == uid;
    }

    // Apresentações em vídeo de cada Instrutor/Professor/Mestre: pasta
    // apresentacoes/<uid da pessoa>/. Quem envia: a própria pessoa (cordão de
    // Instrutor para cima ou papel mestre/instrutor), o Admin, o Fundador e o
    // responsável do núcleo onde ela treina. Vídeo até 20 MB. Logado assiste.
    match /apresentacoes/{uid}/{arquivo} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
                   && request.resource.contentType.matches('video/.*')
                   && request.resource.size < 20 * 1024 * 1024
                   && podeMexerNaApresentacao(uid);
      allow delete: if request.auth != null && podeMexerNaApresentacao(uid);
    }

    // Tudo o que não está listado acima: fechado.
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
